# 算法
## 找出数组的出现最多的数字
- 使用哈希表（字典）
- 遍历数组，使用哈希表（在JavaScript中是对象或Map）记录每个数字出现的次数。
遍历哈希表，找出出现次数最多的数字。
```javascript
function findMostFrequent(arr) {
    let counts = {};
    let maxCount = 0;
    let mostFrequent;

    for(let num of arr) {
        counts[num] = (counts[num] || 0) + 1;
        if(counts[num] > maxCount) {
            maxCount = counts[num];
            mostFrequent = num;
        }
    }

    return mostFrequent;
}

let arr = [2, 2, 1, 1, 1, 2, 2];
console.log(findMostFrequent(arr)); // 输出：2
```
- 排序
    - 先将数组排序。
    - 遍历排序后的数组，计数连续相同的数字，比较并记录最大计数值及其对应的数字。
```javascript
function findMostFrequentBySort(arr) {
    arr.sort((a, b) => a - b);
    let maxCount = 1;
    let currentCount = 1;
    let mostFrequent = arr[0];

    for(let i = 1; i < arr.length; i++) {
        if(arr[i] === arr[i - 1]) {
            currentCount++;
        } else {
            if(currentCount > maxCount) {
                maxCount = currentCount;
                mostFrequent = arr[i - 1];
            }
            currentCount = 1;
        }
    }

    // 检查最后一个元素的计数
    if(currentCount > maxCount) {
        mostFrequent = arr[arr.length - 1];
    }

    return mostFrequent;
}

let arr = [2, 2, 1, 1, 1, 2, 2];
console.log(findMostFrequentBySort(arr)); // 输出：2
```

### 哈希表记录法
``` js
function isUnique(s) {
    const seen = {};
    for (const char of s) {
        if (seen[char]) return false;
        seen[char] = true;
    }
    return true;
}
// 时间复杂度O(n)，空间复杂度O(n)
```
### Set去重

## 判断是否为变位词（重新排列字符串）

### 方法1：排序比较法
```javascript
function isAnagram(s1, s2) {
    // 长度不同直接返回false
    if (s1.length !== s2.length) return false;
    
    // 排序后比较字符串
    return [...s1].sort().join('') === [...s2].sort().join('');
}
// 时间复杂度：O(n log n) 空间复杂度：O(n)
```

### 方法2：哈希表计数法
```javascript
function isAnagram(s1, s2) {
    if (s1.length !== s2.length) return false;
    
    const charCount = new Map();
    
    // 统计s1字符出现次数
    for (const char of s1) {
        charCount.set(char, (charCount.get(char) || 0) + 1);
    }
    
    // 验证s2字符
    for (const char of s2) {
        if (!charCount.has(char)) return false;
        charCount.set(char, charCount.get(char) - 1);
        if (charCount.get(char) === 0) charCount.delete(char);
    }
    
    return charCount.size === 0;
}
// 时间复杂度：O(n) 空间复杂度：O(1)（因为字符集固定）
```

### 方法3：数组计数法（最优解）
```javascript
function isAnagram(s1, s2) {
    if (s1.length !== s2.length) return false;
    
    const count = new Array(26).fill(0);
    const base = 'a'.charCodeAt(0);
    
    // 同时统计两个字符串
    for (let i = 0; i < s1.length; i++) {
        count[s1.charCodeAt(i) - base]++; // s1字符计数加1
        count[s2.charCodeAt(i) - base]--; // s2字符计数减1
    }
    
    // 检查所有计数是否为0
    return count.every(c => c === 0);
}
// 时间复杂度：O(n) 空间复杂度：O(1)
```

### 方法对比
| 方法         | 时间复杂度 | 空间复杂度 | 适用场景               |
|------------|----------|--------|--------------------|
| 排序比较法     | O(n logn)| O(n)   | 快速实现、代码简洁          |
| 哈希表计数法    | O(n)     | O(1)   | 通用字符集（如包含Unicode） |
| 数组计数法     | O(n)     | O(1)   | 仅限小写字母场景（最优解）     |

测试用例：
```javascript
console.log(isAnagram("abc", "bca")); // true
console.log(isAnagram("abc", "bad")); // false
console.log(isAnagram("anagram", "nagaram")); // true
```

## 数组原型对象的最后一个元素
- 编写一段代码实现一个数组方法，使任何数组都可以调用 array.last() 方法，这个方法将返回数组最后一个元素。如果数组中没有元素，则返回 -1 。
- 算法步骤
  - 在名为 last 的数组原型上定义一个新方法。
  - 在这个方法中，检查数组是否为空。如果是，返回 -1。
  - 如果数组不为空，则返回数组的最后一个元素。最后一个元素可以通过以下方式访问：this[this.length - 1]。
  
方式一：常规 if 检查
```js
Array.prototype.last = function() {
  if (this.length === 0) {
    return -1;
  }

  return this[this.length - 1];
}

```
方式二：三元运算符
```js
Array.prototype.last = function() {
 return this.length === 0 ? -1 ：this[this.length -1]
}
```
方式三：合并运算符
```js
Array.prototype.last = function() {
 return this[this.length -1] ?? -1
}
```
方式四：pop方法
```js
Array.prototype.last = function() {
  let val = this.pop();
  return val !== undefined ? val : -1;
}
```
## 根据ID合并数组
现给定两个数组 arr1 和 arr2 ，返回一个新的数组 joinedArray 。两个输入数组中的每个对象都包含一个 id 字段。

joinedArray 是一个通过 id 将 arr1 和 arr2 连接而成的数组。joinedArray 的长度应为唯一值 id 的长度。返回的数组应按 id 升序 排序。

如果一个 id 存在于一个数组中但不存在于另一个数组中，则该对象应包含在结果数组中且不进行修改。

如果两个对象共享一个 id ，则它们的属性应进行合并：

如果一个键只存在于一个对象中，则该键值对应该包含在对象中。
如果一个键在两个对象中都包含，则 arr2 中的值应覆盖 arr1 中的值。
 

示例 1：

输入：
arr1 = [
    {"id": 1, "x": 1},
    {"id": 2, "x": 9}
], 
arr2 = [
    {"id": 3, "x": 5}
]
输出：
[
    {"id": 1, "x": 1},
    {"id": 2, "x": 9},
    {"id": 3, "x": 5}
]
解释：没有共同的 id，因此将 arr1 与 arr2 简单地连接起来。
示例 2：

输入：
arr1 = [
    {"id": 1, "x": 2, "y": 3},
    {"id": 2, "x": 3, "y": 6}
], 
arr2 = [
    {"id": 2, "x": 10, "y": 20},
    {"id": 3, "x": 0, "y": 0}
]
输出：
[
    {"id": 1, "x": 2, "y": 3},
    {"id": 2, "x": 10, "y": 20},
    {"id": 3, "x": 0, "y": 0}
]
解释：id 为 1 和 id 为 3 的对象在结果数组中保持不变。id 为 2 的两个对象合并在一起。arr2 中的键覆盖 arr1 中的值。
示例 3：

输入：
arr1 = [
    {"id": 1, "b": {"b": 94},"v": [4, 3], "y": 48}
]
arr2 = [
    {"id": 1, "b": {"c": 84}, "v": [1, 3]}
]
输出： [
    {"id": 1, "b": {"c": 84}, "v": [1, 3], "y": 48}
]
解释：具有 id 为 1 的对象合并在一起。对于键 "b" 和 "v" ，使用 arr2 中的值。由于键 "y" 只存在于 arr1 中，因此取 arr1 的值。
- 方式一：
```js
var join = function(arr1, arr2) {
    // 数据存放位置
    const datas = new Map();

    for (let item of arr1) {
        update(item, datas);
    }
    for (let item of arr2) {
        update(item, datas);
    }

    // 题目要求要按id排序输出，但是实际用例没有对应打乱的用例
    return [...datas.values()].sort((a,b) => a.id - b.id);
};

var update = function(change, datas) {
    // 获取旧数据，如果没有创建一个
    let target = {};
    if (datas.has(change.id)) target = datas.get(change.id);
    else datas.set(change.id, target);
    // 把key提出出来更新
    for (let key of Object.keys(change)) {
        target[key] = change[key];
    }
}
```

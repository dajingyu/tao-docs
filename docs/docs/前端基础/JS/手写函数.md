## 防抖
```js
function debounce(func, wait) {
    let timeoutId;
    // 返回一个新的函数，该函数在内部处理防抖逻辑
    return function(...args) {
        // 清除之前的定时器，防止函数被提前执行
        clearTimeout(timeoutId);
        // 设置新的定时器，若在wait毫秒内再次调用，将重新计时
        timeoutId = setTimeout(() => {
            func.apply(this, args); // 使用apply确保函数上下文和参数正确
        }, wait);
    };
}
```
## 节流
```js
function throttle(func, wait) {
    let lastExecTime = 0; // 上次执行时间
    // 返回一个新的函数，该函数在内部处理节流逻辑
    return function(...args) {
        const now = new Date().getTime(); // 获取当前时间戳
        // 计算距离上次执行已经过去的时间
        const elapsed = now - lastExecTime;
        // 如果到了执行时间间隔，则执行函数
        if (elapsed >= wait) {
            func.apply(this, args); // 使用apply确保函数上下文和参数正确
            lastExecTime = now; // 更新上次执行时间为当前时间
        }
    };
}
```
## 深拷贝

深拷贝是指创建一个完全独立的对象副本，修改新对象不会影响原对象。

### 方法1：JSON.stringify() + JSON.parse()（最常用）

```javascript
function deepCloneByJSON(obj) {
    return JSON.parse(JSON.stringify(obj));
}

// 使用示例
const original = {
    name: 'Alice',
    age: 30,
    hobbies: ['reading', 'coding'],
    address: {
        city: 'Beijing',
        country: 'China'
    }
};

const cloned = deepCloneByJSON(original);
cloned.name = 'Bob';
cloned.hobbies.push('swimming');
console.log(original.name);        // 'Alice'（未改变）
console.log(original.hobbies);     // ['reading', 'coding']（未改变）
```

#### 优点
- ✅ **简单易用**：代码简洁，一行搞定
- ✅ **性能较好**：对于简单对象，性能表现不错
- ✅ **无需额外依赖**：原生 API，浏览器支持好

#### 缺点
- ❌ **无法处理函数**：函数会被忽略
  ```javascript
  const obj = { fn: function() {} };
  deepCloneByJSON(obj); // { }（函数丢失）
  ```
- ❌ **无法处理 undefined**：undefined 会被忽略
  ```javascript
  const obj = { a: undefined, b: null };
  deepCloneByJSON(obj); // { b: null }（undefined 丢失）
  ```
- ❌ **无法处理 Symbol**：Symbol 属性会被忽略
  ```javascript
  const obj = { [Symbol('key')]: 'value' };
  deepCloneByJSON(obj); // { }（Symbol 丢失）
  ```
- ❌ **无法处理循环引用**：会报错
  ```javascript
  const obj = { a: 1 };
  obj.self = obj;
  deepCloneByJSON(obj); // TypeError: Converting circular structure to JSON
  ```
- ❌ **无法处理 Date 对象**：会变成字符串
  ```javascript
  const obj = { date: new Date() };
  const cloned = deepCloneByJSON(obj);
  console.log(cloned.date); // "2024-01-01T00:00:00.000Z"（字符串，不是 Date 对象）
  console.log(cloned.date instanceof Date); // false
  ```
- ❌ **无法处理 RegExp**：会变成空对象
  ```javascript
  const obj = { regex: /test/g };
  deepCloneByJSON(obj); // { regex: {} }
  ```
- ❌ **无法处理 Map/Set**：会变成空对象
  ```javascript
  const obj = { map: new Map([['a', 1]]) };
  deepCloneByJSON(obj); // { map: {} }
  ```
- ❌ **无法处理 BigInt**：会报错
  ```javascript
  const obj = { big: 9007199254740991n };
  deepCloneByJSON(obj); // TypeError: Do not know how to serialize a BigInt
  ```

#### 适用场景
- ✅ 纯数据对象（不包含函数、Date、RegExp 等）
- ✅ 简单的配置对象
- ✅ 需要快速实现的场景
- ✅ 数据可序列化的场景

### 方法2：递归实现（完整版）

```javascript
function deepClone(obj, hash = new WeakMap()) {
    // 基本类型直接返回
    if (obj === null || typeof obj !== 'object') {
        return obj;
    }
    
    // 处理 Date
    if (obj instanceof Date) {
        return new Date(obj);
    }
    
    // 处理 RegExp
    if (obj instanceof RegExp) {
        return new RegExp(obj);
    }
    
    // 处理循环引用
    if (hash.has(obj)) {
        return hash.get(obj);
    }
    
    // 处理数组
    if (Array.isArray(obj)) {
        const cloneArr = [];
        hash.set(obj, cloneArr);
        obj.forEach((item, index) => {
            cloneArr[index] = deepClone(item, hash);
        });
        return cloneArr;
    }
    
    // 处理对象
    const cloneObj = {};
    hash.set(obj, cloneObj);
    
    // 处理普通属性
    for (let key in obj) {
        if (obj.hasOwnProperty(key)) {
            cloneObj[key] = deepClone(obj[key], hash);
        }
    }
    
    // 处理 Symbol 属性
    const symbolKeys = Object.getOwnPropertySymbols(obj);
    symbolKeys.forEach(symKey => {
        cloneObj[symKey] = deepClone(obj[symKey], hash);
    });
    
    return cloneObj;
}

// 使用示例
const original = {
    name: 'Alice',
    date: new Date(),
    regex: /test/g,
    fn: function() { return 'test'; },
    symbolKey: Symbol('key'),
    [Symbol('hidden')]: 'hidden value',
    arr: [1, 2, { nested: 'value' }]
};

const cloned = deepClone(original);
console.log(cloned.date instanceof Date);     // true
console.log(cloned.regex instanceof RegExp); // true
console.log(typeof cloned.fn);               // 'function'
```

#### 优点
- ✅ **功能完整**：可以处理函数、Date、RegExp、Symbol 等
- ✅ **支持循环引用**：使用 WeakMap 处理循环引用
- ✅ **类型保持**：保持原始数据类型

#### 缺点
- ❌ **代码复杂**：需要处理各种边界情况
- ❌ **性能较差**：递归调用，对于大对象可能较慢
- ❌ **无法处理 Map/Set**：需要额外处理
- ❌ **无法处理 BigInt**：需要特殊处理

### 完整示例：处理各种类型的深拷贝

```javascript
function comprehensiveDeepClone(obj, hash = new WeakMap()) {
    // 基本类型
    if (obj === null || typeof obj !== 'object') {
        return obj;
    }
    
    // 循环引用
    if (hash.has(obj)) {
        return hash.get(obj);
    }
    
    // Date
    if (obj instanceof Date) {
        return new Date(obj.getTime());
    }
    
    // RegExp
    if (obj instanceof RegExp) {
        return new RegExp(obj);
    }
    
    // Map
    if (obj instanceof Map) {
        const cloneMap = new Map();
        hash.set(obj, cloneMap);
        obj.forEach((value, key) => {
            cloneMap.set(key, comprehensiveDeepClone(value, hash));
        });
        return cloneMap;
    }
    
    // Set
    if (obj instanceof Set) {
        const cloneSet = new Set();
        hash.set(obj, cloneSet);
        obj.forEach(value => {
            cloneSet.add(comprehensiveDeepClone(value, hash));
        });
        return cloneSet;
    }
    
    // Array
    if (Array.isArray(obj)) {
        const cloneArr = [];
        hash.set(obj, cloneArr);
        obj.forEach((item, index) => {
            cloneArr[index] = comprehensiveDeepClone(item, hash);
        });
        return cloneArr;
    }
    
    // Object
    const cloneObj = {};
    hash.set(obj, cloneObj);
    
    // 普通属性
    for (let key in obj) {
        if (obj.hasOwnProperty(key)) {
            cloneObj[key] = comprehensiveDeepClone(obj[key], hash);
        }
    }
    
    // Symbol 属性
    const symbolKeys = Object.getOwnPropertySymbols(obj);
    symbolKeys.forEach(symKey => {
        cloneObj[symKey] = comprehensiveDeepClone(obj[symKey], hash);
    });
    
    return cloneObj;
}
```

## pormise
```js
class MyPromise {
    constructor(executor) {
        this.state = 'pending';
        this.value = undefined;
        this.reason = undefined;
        this.onFulfilledCallbacks = [];
        this.onRejectedCallbacks = [];

        try {
            executor(this.resolve.bind(this), this.reject.bind(this));
        } catch (error) {
            this.reject(error);
        }
    }

    resolve(value) {
        if (this.state === 'pending') {
            this.state = 'fulfilled';
            this.value = value;
            this.onFulfilledCallbacks.forEach(fn => fn(value));
        }
    }

    reject(reason) {
        if (this.state === 'pending') {
            this.state = 'rejected';
            this.reason = reason;
            this.onRejectedCallbacks.forEach(fn => fn(reason));
        }
    }

    then(onFulfilled, onRejected) {
        onFulfilled = typeof onFulfilled === 'function' ? onFulfilled : value => value;
        onRejected = typeof onRejected === 'function' ? onRejected : reason => { throw reason };

        let promise2 = new MyPromise((resolve, reject) => {
            switch (this.state) {
                case 'fulfilled':
                    setTimeout(() => {
                        try {
                            let x = onFulfilled(this.value);
                            resolvePromise(promise2, x, resolve, reject);
                        } catch (e) {
                            reject(e);
                        }
                    }, 0);
                    break;
                case 'rejected':
                    setTimeout(() => {
                        try {
                            let x = onRejected(this.reason);
                            resolvePromise(promise2, x, resolve, reject);
                        } catch (e) {
                            reject(e);
                        }
                    }, 0);
                    break;
                default:
                    this.onFulfilledCallbacks.push(() => {
                        setTimeout(() => {
                            try {
                                let x = onFulfilled(this.value);
                                resolvePromise(promise2, x, resolve, reject);
                            } catch (e) {
                                reject(e);
                            }
                        }, 0);
                    });
                    this.onRejectedCallbacks.push(() => {
                        setTimeout(() => {
                            try {
                                let x = onRejected(this.reason);
                                resolvePromise(promise2, x, resolve, reject);
                            } catch (e) {
                                reject(e);
                            }
                        }, 0);
                    });
                    break;
            }
        });

        return promise2;
    }

    catch(onRejected) {
        return this.then(null, onRejected);
    }
}

// 辅助函数，解决Promise的链式调用中返回值的处理问题
function resolvePromise(promise2, x, resolve, reject) {
    if (x === promise2) {
        reject(new TypeError('Chaining cycle detected for promise #<Promise>'));
    }

    if ((typeof x === 'object' && x !== null) || typeof x === 'function') {
        let called = false;
        try {
            let then = x.then;
            if (typeof then === 'function') {
                then.call(x, (y) => {
                    if (called) return;
                    called = true;
                    resolvePromise(promise2, y, resolve, reject);
                }, (r) => {
                    if (called) return;
                    called = true;
                    reject(r);
                });
            } else {
                resolve(x);
            }
        } catch (e) {
            if (called) return;
            called = true;
            reject(e);
        }
    } else {
        resolve(x);
    }
}
```

function asyncToGenerator(generatorFunc) {
    return function() {
        const gen = generatorFunc.apply(this, arguments);
        return new Promise((resolve, reject) => {
            function step(key, arg) {
                let result;
                try {
                    result = gen[key](arg);
                } catch (error) {
                    return reject(error);
                }
                if (result.done) {
                    return resolve(result.value);
                } else {
                    return Promise.resolve(result.value).then(
                        val => step("next", val),
                        err => step("throw", err)
                    );
                }
            }
            step("next");
        });
    };
}

        }, 1000);
    });
}

## 手写事件发布、订阅
``` js
class EventEmitter {
    constructor() {
        this.events = {}; // 用于存储事件类型及其对应的回调函数列表
    }

    /**
     * 订阅事件
     * @param {string} eventName 事件名称
     * @param {Function} listener 回调函数
     */
    on(eventName, listener) {
        if (!this.events[eventName]) {
            this.events[eventName] = [];
        }
        this.events[eventName].push(listener);
    }

    /**
     * 发布事件
     * @param {string} eventName 事件名称
     * @param  {...any} args 传递给监听器的参数
     */
    emit(eventName, ...args) {
        const listeners = this.events[eventName];
        if (listeners) {
            listeners.forEach(listener => listener(...args));
        }
    }

    /**
     * 取消订阅
     * @param {string} eventName 事件名称
     * @param {Function} [listener] 特定的回调函数，如果不传则移除所有该事件的监听器
     */
    off(eventName, listener) {
        if (this.events[eventName]) {
            if (listener) {
                // 移除特定的监听器
                this.events[eventName] = this.events[eventName].filter(l => l !== listener);
            } else {
                // 如果没有指定监听器，则清空该事件的所有监听器
                delete this.events[eventName];
            }
        }
    }

    /**
     * 一次性订阅，事件触发后自动移除监听器
     * @param {string} eventName 事件名称
     * @param {Function} listener 回调函数
     */
    once(eventName, listener) {
        const wrappedListener = (...args) => {
            listener(...args);
            this.off(eventName, wrappedListener);
        };
        this.on(eventName, wrappedListener);
    }
}

// 使用示例
const emitter = new EventEmitter();

// 订阅事件
emitter.on('greet', name => console.log(`Hello, ${name}!`));

// 发布事件
emitter.emit('greet', 'Alice');

// 取消订阅
// emitter.off('greet');

// 一次性订阅
emitter.once('goodbye', () => console.log('Goodbye!'));
emitter.emit('goodbye');
emitter.emit('goodbye'); // 这次不会触发，因为上一次调用后已经移除了监听器
```

## sort实现原理
JavaScript 的 `Array.prototype.sort` 在不同引擎中的实现不同，V8 引擎的演进最具代表性：

### 核心实现策略
1. **短数组（长度 ≤10）**：使用插入排序（稳定且对小数据高效）
2. **长数组**：
   - 7.0 之前：快速排序（不稳定的通用排序）
   - 7.0 之后：TimSort（Python 使用的混合排序算法，稳定且擅长处理真实数据）

### 关键处理逻辑
```js
// 简化版实现逻辑示意
function mySort(array, comparefn) {
    // 处理 compare 函数：当没有提供比较函数时，默认将元素转为字符串比较
    const defaultCompare = (a, b) => {
        a = String(a);  // 将元素转为字符串
        b = String(b);
        return a < b ? -1 : a > b ? 1 : 0; // 按字典序比较
    };
    const comparator = comparefn || defaultCompare; // 优先使用自定义比较函数

    // 插入排序（适合短数组，稳定算法）
    function insertionSort(arr) {
        // 从第二个元素开始遍历（i=1）
        for (let i = 1; i < arr.length; i++) {
            const temp = arr[i]; // 当前待插入元素
            let j = i - 1;       // 从已排序部分的末尾开始比较
            
            // 寻找插入位置：当已排序元素 > 待插入元素时，向后移动元素
            while (j >= 0 && comparator(arr[j], temp) > 0) {
                arr[j + 1] = arr[j]; // 元素后移
                j--;                // 向前比较
            }
            arr[j + 1] = temp; // 插入到正确位置
        }
    }

    // 快速排序基础实现（适合长数组，非稳定算法）
    function quickSort(arr) {
        if (arr.length <= 1) return arr; // 递归终止条件
        
        const pivot = arr[0];  // 选择第一个元素为基准（实际实现有更优的基准选择）
        const left = [];       // 存储小于基准的元素
        const right = [];      // 存储大于等于基准的元素
        
        // 分区操作（从第二个元素开始比较）
        for (let i = 1; i < arr.length; i++) {
            comparator(arr[i], pivot) < 0 
                ? left.push(arr[i])   // 小于基准放左边
                : right.push(arr[i]); // 否则放右边
        }
        // 递归排序并合并结果
        return [...quickSort(left), pivot, ...quickSort(right)];
    }

    // 根据数组长度选择排序算法
    return array.length <= 10 
        ? insertionSort(array)  // 短数组使用插入排序
        : quickSort(array);     // 长数组使用快速排序
}
```

### 注意事项
1. **稳定性**：现代浏览器（Chrome 70+）使用 TimSort 实现稳定排序
   - 稳定排序能保持相等元素的原始顺序
2. **类型转换**：默认比较时会将元素转为字符串（`[1, 2, 10].sort()` → `[1, 10, 2]`）
   - 数值排序应使用 `arr.sort((a,b) => a - b)`
3. **性能优化**：实际实现包含更多优化：
   - 三路快速排序（处理大量重复元素）
   - 插入排序阈值动态调整（根据数组特征）
   - 避免最坏情况下的快速排序性能问题
4. **稀疏数组**：会跳过空元素（`[1,,3].sort()` → `[1, 3, empty]`）
   - undefined 元素会被排到数组末尾

function myStringify(value) {
    // 循环引用检测
    const seen = new WeakSet();

    function stringifyHelper(val) {
        // 处理基本类型
        if (val === null) return 'null';
        if (typeof val === 'number') {
            // 处理特殊数值
            return isNaN(val) || !isFinite(val) ? 'null' : val.toString();
        }
        if (typeof val === 'boolean') return val.toString();
        if (typeof val === 'string') return `"${val.replace(/"/g, '\\"')}"`;
        if (typeof val === 'undefined') return undefined;
        
        // 处理对象引用
        if (typeof val === 'object') {
            if (seen.has(val)) throw new TypeError('循环引用');
            seen.add(val);

            // 处理数组
            if (Array.isArray(val)) {
                const items = val.map(item => 
                    typeof item === 'undefined' || typeof item === 'function' 
                        ? 'null' 
                        : stringifyHelper(item)
                );
                seen.delete(val);
                return `[${items.join(',')}]`;
            }

            // 处理普通对象
            const entries = [];
            for (const key in val) {
                if (val.hasOwnProperty(key)) {
                    const item = val[key];
                    if (typeof item !== 'undefined' && typeof item !== 'function') {
                        entries.push(`"${key}":${stringifyHelper(item)}`);
                    }
                }
            }
            seen.delete(val);
            return `{${entries.join(',')}}`;
        }

        // 其他类型处理
        return 'null';
    }

    try {
        return stringifyHelper(value);
    } catch (e) {
        console.error('序列化失败:', e);
        return 'null';
    }
}

## 手写JSON.parse

### 核心实现（安全版）
```javascript
function myParse(jsonStr) {
    let index = 0;
    const str = jsonStr.trim();

    // 错误处理
    function syntaxError(msg) {
        throw new SyntaxError(`JSON解析错误: ${msg} at position ${index}`);
    }

    // 主解析函数
    function parseValue() {
        const char = str[index];
        if (char === '{') return parseObject();
        if (char === '[') return parseArray();
        if (char === '"') return parseString();
        if (char === 't' && str.substr(index, 4) === 'true') {
            index += 4; return true;
        }
        if (char === 'f' && str.substr(index, 5) === 'false') {
            index += 5; return false;
        }
        if (char === 'n' && str.substr(index, 4) === 'null') {
            index += 4; return null;
        }
        if (/[-0-9]/.test(char)) return parseNumber();
        syntaxError('意外的标记');
    }

    // 解析对象
    function parseObject() {
        index++; // 跳过 '{'
        const obj = {};
        while (str[index] !== '}') {
            if (index >= str.length) syntaxError('未闭合的对象');
            const key = parseString();
            index = str.indexOf(':', index) + 1;
            const value = parseValue();
            obj[key] = value;
            if (str[index] === ',') index++;
        }
        index++; // 跳过 '}'
        return obj;
    }

    // 解析数组
    function parseArray() {
        index++; // 跳过 '['
        const arr = [];
        while (str[index] !== ']') {
            if (index >= str.length) syntaxError('未闭合的数组');
            arr.push(parseValue());
            if (str[index] === ',') index++;
        }
        index++; // 跳过 ']'
        return arr;
    }

    // 解析字符串
    function parseString() {
        index++; // 跳过起始引号
        let result = '';
        while (str[index] !== '"') {
            if (str[index] === '\\') {
                index++;
                switch (str[index]) {
                    case '"': result += '"'; break;
                    case '\\': result += '\\'; break;
                    case '/': result += '/'; break;
                    case 'b': result += '\b'; break;
                    case 'f': result += '\f'; break;
                    case 'n': result += '\n'; break;
                    case 'r': result += '\r'; break;
                    case 't': result += '\t'; break;
                    case 'u': 
                        const hex = str.substr(index+1, 4);
                        result += String.fromCharCode(parseInt(hex, 16));
                        index +=4;
                        break;
                    default: syntaxError('无效的转义字符');
                }
            } else {
                result += str[index];
            }
            index++;
        }
        index++; // 跳过结束引号
        return result;
    }

    // 解析数字
    function parseNumber() {
        const start = index;
        while (/[-+0-9.eE]/.test(str[index])) index++;
        const numStr = str.slice(start, index);
        const num = parseFloat(numStr);
        if (isNaN(num)) syntaxError('无效数字格式');
        return num;
    }

    return parseValue();
}
```

### 注意事项
1. 简化实现限制：
   - 不支持尾逗号
   - 未完整实现Unicode转义
   - 未处理深层嵌套（超过调用栈限制）
   - 未实现完整的数字格式校验
2. 安全特性：
   - 避免使用eval
   - 基本语法校验
   - 防止原型污染

  });

## 手写函数柯里化

### 基础版本
```javascript
function curry(fn) {
  return function curried(...args) {
    if (args.length >= fn.length) {
      return fn.apply(this, args);
    } else {
      return function(...nextArgs) {
        return curried.apply(this, args.concat(nextArgs));
      };
    }
  };
}

// 使用
function add(a, b, c) {
  return a + b + c;
}

const curriedAdd = curry(add);
curriedAdd(1)(2)(3);    // 6
curriedAdd(1, 2)(3);    // 6
curriedAdd(1)(2, 3);    // 6
```

### 支持占位符版本
```javascript
function curryWithPlaceholder(fn) {
  return function curried(...args) {
    // 过滤占位符
    const validArgs = args.filter(arg => arg !== curryWithPlaceholder.placeholder);
    
    if (validArgs.length >= fn.length) {
      return fn.apply(this, args);
    }
    
    return function(...nextArgs) {
      // 替换占位符
      const newArgs = args.map(arg => 
        arg === curryWithPlaceholder.placeholder && nextArgs.length > 0
          ? nextArgs.shift()
          : arg
      );
      return curried.apply(this, newArgs.concat(nextArgs));
    };
  };
}

curryWithPlaceholder.placeholder = Symbol('placeholder');

// 使用

## 手写 Array.prototype.reduce

### 核心实现
```javascript
Array.prototype.myReduce = function(callback, initialValue) {
  if (this == null) {
    throw new TypeError('Array.prototype.reduce called on null or undefined');
  }
  
  if (typeof callback !== 'function') {
    throw new TypeError(callback + ' is not a function');
  }
  
  const len = this.length;
  let accumulator = initialValue;
  let startIndex = 0;
  
  if (initialValue === undefined) {
    if (len === 0) {
      throw new TypeError('Reduce of empty array with no initial value');
    }
    accumulator = this[0];
    startIndex = 1;
  }
  
  for (let i = startIndex; i < len; i++) {
    accumulator = callback(accumulator, this[i], i, this);
  }
  
  return accumulator;
};
```

// 使用
[1, 2, 3, 4].myReduce((acc, cur) => acc + cur, 0); // 10
[1, 2, 3, 4].myReduce((acc, cur) => acc + cur);    // 10
